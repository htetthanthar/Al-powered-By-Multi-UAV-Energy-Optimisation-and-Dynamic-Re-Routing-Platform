\begin{frame}{Problem Statement: The Aerial Last-Mile Reality}
  \begin{columns}
    \column{0.55\textwidth}
    \textbf{Why Classical Heuristics Fail in Real Airspace:}
    \begin{itemize}
      \item Classical Vehicle Routing (CVRP/2-Opt) assumes \textbf{static linear battery drain} ($0.8 \times \text{distance}$).
      \item Real multi-rotor power draw is \textbf{non-linear} and dictated by aerodynamics:
      $$P_{\text{total}} = P_{\text{induced}} + P_{\text{profile}} + P_{\text{parasite}}$$
      \item A $20\text{ km/h}$ localized headwind increases continuous power draw by over $40\%$, causing unexpected mid-flight stranding.
    \end{itemize}
    \column{0.45\textwidth}
    \begin{alertblock}{Operational Boundaries}
      \begin{itemize}
        \item \textbf{Payload Limit:} $\le 2.0\text{ kg}$
        \item \textbf{Battery Duration:} $\le 55.0\text{ min}$
        \item \textbf{Cruise Speed:} $65.0\text{ km/h}$
        \item \textbf{Weather Safety:} Wind $\le 30\text{ km/h}$
      \end{itemize}
    \end{alertblock}
  \end{columns}
\end{frame}

\begin{frame}{End-to-End System Architecture}
  \centering
  \begin{block}{5-Tier Integrated Pipeline}
    \begin{enumerate}
      \item \textbf{Geofence Triage:} Ingests addresses, extracts No-Fly Zones (NFZs), and diverts restricted stops to ground fleets.
      \item \textbf{Deep Learning Surrogates:}
      \begin{itemize}
        \item \textit{Weather Bi-LSTM:} Predicts localized wind vectors $(\hat{u}, \hat{v})$ and precipitation flags ($\text{MAE} = 0.40\text{ m/s}$).
        \item \textit{Energy MLP:} Predicts non-linear flight power draw ($R^2 = 0.9782, \text{RMSE} = 5.84\text{ W}$).
      \end{itemize}
      \item \textbf{Combinatorial Routing Engine:} Graph Attention Network (GAT) with dynamic logit masking.
      \item \textbf{Dynamic In-Flight Dispatch:} FSM monitoring sensor EKF and executing automated $180\text{s}$ hover holds or RTB.
      \item \textbf{Hardware \& Telemetry:} Native ROS 2 / MAVLink offboard loop and DuckDB analytical data lake.
    \end{enumerate}
  \end{block}
\end{frame}

\begin{frame}{Deep Learning Surrogate Formulation}
  \begin{columns}
    \column{0.5\textwidth}
    \textbf{MLP Power Surrogate:}
    \begin{itemize}
      \item Ingests 5 physical variables:
      $$\mathbf{x} = [v_{\text{ground}}, m_{\text{payload}}, w_{\text{speed}}, \theta_{\text{wind}}, \Delta h]^\top$$
      \item Evaluates candidate edges in $\mathbf{0.08\text{ ms}}$ on Jetson ONNX runtime.
    \end{itemize}
    \column{0.5\textwidth}
    \textbf{Graph Attention Logit Masking:}
    \begin{itemize}
      \item Enforces hard constraints directly inside decoder:
      $$p(\pi_t = j) = \frac{\exp(u_{t,j} + \mathcal{M}_{t,j})}{\sum_k \exp(u_{t,k} + \mathcal{M}_{t,k})}$$
      $$\mathcal{M}_{t,j} = \begin{cases} 0 & \text{if feasible} \\ -\infty & \text{if payload/energy violated} \end{cases}$$
    \end{itemize}
  \end{columns}
\end{frame}
